# =========================================================================
#  ONLYSTYLE 业务系统 · 部署配置
#  第一次双击 deploy.bat 会把本模板复制成 deploy.config.txt，
#  然后只需要填 SERVER_IP 这一行，其余保持默认即可。
#  注意：= 两边不要加空格；# 开头的行是注释。
#
#  现网拓扑（先看懂再填）：
#    本机(Windows)  --构建+打包+scp-->  应用服务器(pm2 双进程)
#                                        onlystyle-api :3100
#                                        onlystyle-web :8085
#                                              ▲
#                                              │ upstream 反代
#                                        代理服务器(Nginx)
#                                  www.域名 → 8085 / bos.域名 → 3100
# =========================================================================

# 【必填】应用服务器 IP（云控制台实例列表里能看到）
SERVER_IP=

# 【一般不用改】SSH 端口，默认 22
SSH_PORT=22

# 【一般不用改】登录用户名。腾讯云/阿里云轻量的 root 直接就填 root
SSH_USER=root

# 【通常不用改】服务器上的安装根目录（与 ecosystem.config.js 里的绝对路径一致）
APP_DIR=/opt/business-os

# 【建议保持默认】域名。用于打印外网入口、以及代理机 nginx 的 server_name。
#   官网 = www.<DOMAIN>，后台 = bos.<DOMAIN>
DOMAIN=onlystyle.com.cn

# -------------------------------------------------------------------------
#  说明：以下旧配置项已废弃（现网不再适用），留此备忘避免误填：
#    WEB_PORT / ADMIN_PORT —— 现网官网固定 8085、后台固定 3100，由
#                             ecosystem.config.js 决定，不在这里配。
#    ADMIN_USER / ADMIN_PASS —— 旧版单机 nginx 的 Basic Auth。现网后台鉴权
#                               走应用层（登录 token + lib/admin-auth.js），
#                               nginx 层不再设口令，所以这两项已删除。
#    USE_HTTPS —— 证书在**代理服务器**上由 certbot 申请，不在应用服务器做。
# -------------------------------------------------------------------------

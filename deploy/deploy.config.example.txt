# =========================================================================
#  ONLYSTYLE 业务系统 · 部署配置
#  第一次双击 deploy.bat 会自动把这个模板复制成 deploy.config.txt
#  然后你只需要改 SERVER_IP 这一行，其余都可以先不管。
#  注意：= 两边不要加空格；# 开头的行是注释。
# =========================================================================

# 【必填】服务器公网 IP（在云控制台的实例列表里能看到）
SERVER_IP=

# 【一般不用改】SSH 端口，默认 22
SSH_PORT=22

# 【一般不用改】登录用户名。腾讯云/阿里云轻量的 root 直接就填 root
SSH_USER=root

# 【可选】官网对外端口。默认 8080（避开需要备案的 80）
WEB_PORT=8080

# 【可选】后台对外端口。默认 8081
ADMIN_PORT=8081

# 【可选】后台访问口令的用户名（会弹一个浏览器原生登录框，先把门锁上）
ADMIN_USER=onlystyle

# 【可选】后台访问口令的密码。
#   留空 = 部署时自动生成一个 14 位随机密码，并在部署结束后打印在屏幕上
ADMIN_PASS=

# 【可选】域名。没备案 / 还没解析，就整个留空，用 IP:端口 访问。
#   填了的话脚本会额外生成 example.com / www.example.com（官网）和
#   admin.example.com（后台）三个站点，前提是域名已经解析到本机 IP。
DOMAIN=

# 【可选】是否申请 HTTPS 证书（Let's Encrypt）。
#   需要 DOMAIN 已填且已解析；大陆节点还必须先完成 ICP 备案，否则会失败。
#   填 1 = 申请，留空 = 不申请
USE_HTTPS=
